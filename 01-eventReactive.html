eventReactive() and actionButton()

These functions are usually employed together:

The action button

actionButton("start", 
             "Press here to do something")

The action button controlling the actions

ui = fluidPage(
  [...]
  actionButton("start",
             "Press here to do something")
)
server = server(input, output) {
  data = eventReactive(input$start,
                    [...]
                       )
}

Action button visible in the UI















server <- function(input, output){
    dataInput <- eventReactive(input$load, {
      if(input$dataset == 1){
        data <- rock
      } else if (input$dataset == 2 ){
        data <- pressure
      } else if (input$dataset == 3) {
        data <- cars
      }
    }
    )
output$graph <- renderPlot({
      data <- dataInput()
      plot(data[, c(1:2)])
    })
output$summary <- renderPrint({
      data <- dataInput()
      summary(data[, c(1:2)])
    })
  }
